// ----------------------------- Helpers ---------------------------------------
f_round_down(float x) => math.floor(x * 100.0 + 1e-10) / 100.0
f_round_up(float x)   => math.ceil(x * 100.0 - 1e-10) / 100.0


f_fee(float q, float px) =>
    float commission = math.min(math.max(0.0035 * q, 0.35), 0.01 * q * px)
    q * px * COST_BPS / 10000.0 + commission


f_floor_qty(float allocation, float px) =>
    int q = na(px) or px <= 0 ? 0 : int(math.max(0, math.floor(allocation / px)))
    while q > 0 and q * px + f_fee(q, px) > allocation
        q -= 1
    q


f_push(array<float> a, float v, int maxn) =>
    array.push(a, v)
    if array.size(a) > maxn
        array.shift(a)


f_back(array<float> a, int n) =>
    int s = array.size(a)
    s > n ? array.get(a, s - 1 - n) : na


f_mean_last(array<float> a, int n) =>
    int s = array.size(a)
    float out = na
    if s >= n
        float sum = 0.0
        bool ok = true
        for k = 0 to n - 1
            float v = array.get(a, s - 1 - k)
            if na(v)
                ok := false
            else
                sum += v
        out := ok ? sum / n : na
    out


f_std_last(array<float> a, int n) =>
    int s = array.size(a)
    float out = na
    if s >= n
        float mu = f_mean_last(a, n)
        if not na(mu)
            float ss = 0.0
            for k = 0 to n - 1
                float v = array.get(a, s - 1 - k)
                ss += math.pow(v - mu, 2)
            out := math.sqrt(ss / n)
    out


f_max_last(array<float> a, int n) =>
    int s = array.size(a)
    float out = na
    if s >= n
        bool ok = true
        float m = -1e100
        for k = 0 to n - 1
            float v = array.get(a, s - 1 - k)
            if na(v)
                ok := false
            else
                m := math.max(m, v)
        out := ok ? m : na
    out


f_min_last(array<float> a, int n) =>
    int s = array.size(a)
    float out = na
    if s >= n
        bool ok = true
        float m = 1e100
        for k = 0 to n - 1
            float v = array.get(a, s - 1 - k)
            if na(v)
                ok := false
            else
                m := math.min(m, v)
        out := ok ? m : na
    out


// Python rma(): SMA seed after n contiguous finite values; any gap resets.
// State array layout: [last, seedSum, count]. Arrays are reference types, so
// this helper mutates state in place and returns only the current output.
f_rma_step(float x, int n, array<float> st) =>
    float last = array.get(st, 0)
    float seed = array.get(st, 1)
    int count = int(array.get(st, 2))
    float out = na
    if na(x)
        last := na
        seed := 0.0
        count := 0
    else if count < n
        seed += x
        count += 1
        if count == n
            last := seed / n
            out := last
    else
        last := (last * (n - 1) + x) / n
        out := last
    array.set(st, 0, last)
    array.set(st, 1, seed)
    array.set(st, 2, count)
    out


// Python ema(): first-finite recursive seed, output only after n observations.
// State array layout: [last, count].
f_ema_step(float x, int n, array<float> st) =>
    float last = array.get(st, 0)
    int count = int(array.get(st, 1))
    float out = na
    if na(x)
        last := na
        count := 0
    else
        last := na(last) ? x : (2.0 / (n + 1.0)) * x + (1.0 - 2.0 / (n + 1.0)) * last
        count += 1
        if count >= n
            out := last
    array.set(st, 0, last)
    array.set(st, 1, count)
    out
